\begin{frame}[t]{Six hyperfine states in a deuterium atom}
\lead{Before ionization, the ground-state atom contains an electron and a spin-1 nucleus}
\begin{columns}[T,onlytextwidth]
\column{0.50\textwidth}
\[
 S=\tfrac12,\quad I=1
 \quad\Longrightarrow\quad (2S+1)(2I+1)=6
\]
\smalltext
At $B=0$, electron and nuclear spins couple:\par
\vspace{1mm}
$F=3/2$: four substates.\quad $F=1/2$: two substates.\par
\vspace{5mm}
\textbf{State labels in the strong-field limit}\par
\vspace{1mm}
\begin{tabularx}{\linewidth}{@{}p{37mm}*{3}{>{\centering\arraybackslash}X}@{}}
\toprule
 & $m_I=+1$ & $m_I=0$ & $m_I=-1$\\
\midrule
$m_S=+1/2$ & $|1\rangle$ & $|2\rangle$ & $|3\rangle$\\
$m_S=-1/2$ & $|6\rangle$ & $|5\rangle$ & $|4\rangle$\\
\bottomrule
\end{tabularx}\par
\vspace{4mm}
At weaker fields, the eigenstates generally mix electron and nuclear spin projections.\par
\column{0.46\textwidth}
% [EN] Original Fig. 2.3, PDF page 47, cropped by Poppler at scale 2000: x=480, y=375, width=690, height=575. / [CN] 原图 2.3，PDF 第 47 页，以 Poppler 按上述像素框裁取，未改动图内内容。
\includegraphics[width=\linewidth,height=83mm,keepaspectratio]{figures/deuterium-hyperfine-states.png}\par
\vspace{2mm}
\smalltext
The static field $B_0$ sets the six energies\par
\vspace{1mm}
$E_1(B_0),E_2(B_0),\ldots,E_6(B_0)$.\par
\vspace{2mm}
Their separations determine the RF resonances.
\end{columns}
\source{Source and original Breit--Rabi diagram: \src{\hyperfineref}{Sabourov, Duke PhD thesis (2006), pp. 22--23, Fig. 2.3 / PDF pp. 46--47}. For the $1s$ electron, $J=S$, so the figure's $m_j$ is the table's $m_S$. Spin projections refer to the local static-field axis.}
\note{\noteone}
\end{frame}

\begin{frame}[t]{RF transitions control the nuclear spin populations}
\smalltext
\setlength{\abovedisplayskip}{3pt}
\setlength{\belowdisplayskip}{6pt}
\vspace{-2mm}
\[
 \mathbf B_{\rm RF}(t)=\mathbf B_1\cos(2\pi\nu t),\qquad
 h\nu=\left|E_j(B_0)-E_i(B_0)\right|
 \quad\text{for an allowed single-photon transition}
\]
\vspace{-2mm}
\begin{columns}[T,onlytextwidth]
\column{0.48\textwidth}
\textbf{WFT = Weak Field Transition}\par
Weak static field: within an $F$ multiplet.
\column{0.48\textwidth}
\textbf{SFT = Strong Field Transition}\par
Transfers between the two hyperfine multiplets.
\end{columns}
\vspace{3mm}
\lead{RF redistributes populations. Sextupoles select which atoms remain in the beam.}
\renewcommand{\arraystretch}{1.12}
\begin{tabularx}{\textwidth}{@{}p{43mm}p{37mm}Y@{}}
\toprule
Stage & States & Role in the ideal $p_{yy}=+1$ recipe\\
\midrule
Sextupole \#1 & $1,2,3$ & Focus the selected electron-spin group.\\
WFT\#1 & $2,3,4$ & Transfer $1\to4$ effectively through intermediate states.\\
Sextupole \#2 & $2,3$ & Reject state 4 and retain the focused states.\\
SFT\#2 & $3,6$ & Transfer $2\to6$, filling the $m_I=+1$ branch.\\
\bottomrule
\end{tabularx}\par
\vspace{2mm}
ECR removes the electron. Equal populations of states 3 and 6 give, ideally:\par
\vspace{1mm}
\( (n_+,n_0,n_-)=(\tfrac12,0,\tfrac12)\quad\Longrightarrow\quad
 p_y=n_+-n_-=0,\qquad p_{yy}=1-3n_0=+1.\)\par
\vspace{2mm}
{\fontsize{13}{16}\selectfont The nuclear populations use the strong-field limit and ideal polarization retention. SFT\#1 and WFT\#2 are off. We retain the vertical source axis without additional Wien-filter rotation.}\par
\source{Sources: \src{\rfprinciple}{Mikirtychyants et al., arXiv:1212.1840, Sec. II G}; \src{\okamura}{Okamura et al., RIKEN APR 28 (1995), Table 1, p. 148}; \src{\sakamoto}{Sakamoto et al., 2016, p. 146}. Historical recipe with ideal state selection, transfer, and ionization.}
\note{\notetwo}
\end{frame}

\begin{frame}[t]{Beam attenuation before F3}
\lead{F3 total beam rate: $\dot N_{\mathrm{F3}}\leq10^7$ particles/s}
{\smalltext Proposed order, subject to beamline and interlock approval:\par}
\vspace{0mm}
\[
\mathrm{SRC}\ \longrightarrow\ \mathrm{polarimeter}\ \longrightarrow\ \mathrm{attenuation}\ \longrightarrow\ \mathrm{F3}\ \longrightarrow\ \mathrm{SAMURAI}
\]
\vspace{1mm}
\begin{columns}[T,onlytextwidth]
\column{0.48\textwidth}
\textbf{ATT: mesh beam attenuator}\par
\vspace{2mm}
\smalltext
Open apertures transmit part of the beam.\par
\vspace{2mm}
Nominal ratios: $1/2$, $1/3$, $1/5$, $1/10$, $1/100$.\par
\vspace{2mm}
Documented locations: C22, S31a and S64, upstream of RRC.\par
\vspace{2mm}
Upstream attenuation also lowers the after-SRC count rate.
\column{0.48\textwidth}
\textbf{Slits: acceptance restriction}\par
\vspace{2mm}
\smalltext
Select position, angle, or momentum.\par
\vspace{2mm}
F2--F3 angular-selection slits are documented.\par
\vspace{2mm}
Reduction depends on the beam distribution and aperture settings.\par
\vspace{2mm}
Assess stability, intercepted beam, and the polarization of the transmitted fraction.
\end{columns}
\vspace{2mm}
{\smalltext\color{Rust}\textbf{Interlock condition:} the 2025 BIS report describes an upstream-ATT check for primary-beam delivery. Downstream slits alone cannot be assumed to satisfy it.}
\source{Sources: \src{\fthreelimit}{BigRIPS F3 instructions}; \src{\attguide}{RIKEN intensity guide, p. 7}; \src{\attlocations}{APR 47 (2014), p. 156}; \src{\slitsource}{APR 52 (2019), p. 130}; \src{\bis}{APR 58 (2025), pp. 98--99}. Confirm the applicable operating mode and current interlock configuration.}
\note{\notefive}
\end{frame}
