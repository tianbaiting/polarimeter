\begin{frame}{Spin-1: 8 general degrees, 5 tensor, 1 prepared}
  \begin{columns}[T,onlytextwidth]
    \column{0.47\textwidth}
      For $w_a\geq0$ and $\sum_a w_a=1$,
      \[
        \rho=\sum_a w_a|\psi_a\rangle\langle\psi_a|.
      \]
      A normalized $3\times3$ Hermitian $\rho$ has
      \[
        \boxed{8=3_{\rm vector}+5_{\rm tensor}}
      \]
      independent real parameters.
    \column{0.49\textwidth}
      An axial tensor prepared about $\hat{\vct n}$ is
      \[
        \Tens_0(q,\hat{\vct n})
        =\frac q2\left(3\hat{\vct n}\hat{\vct n}^{\mathsf T}-I\right).
      \]
      \[
        \hat{\vct n}=\hat{\vct y}:
        \quad \Tens_0=\frac{p_{yy}}2\operatorname{diag}(-1,2,-1),
      \]
      \vspace{-0.45em}
      \[
        \hat{\vct n}=\hat{\vct z}:
        \quad \Tens_0=\frac{p_{zz}}2\operatorname{diag}(-1,-1,2).
      \]
  \end{columns}

  \vspace{0.45em}
  \conclusion{Tensor only: 5 degrees. Fixed preparation axis: one amplitude $q$.}
  \note{[Sources] Ohlsen spin-1 density-matrix convention; DPOLAR Cartesian density-matrix parameterization. pyy and pzz are two preparation axes of the same axial-tensor family.}

% [EN] Separate physical spin evolution from the passive change to the downstream beam basis, and state the rank-preserving scope of linear BMT. / [CN] 将真实自旋演化与下游束流坐标系的被动变换分开，并明确线性 BMT 保持张量阶数的适用范围。
\begin{frame}{Linear BMT preserves vector and tensor ranks}
  \begin{columns}[T,onlytextwidth]
    \column{0.47\textwidth}
      \textbf{1. Magnetic field rotates the spin in the lab}
      \[
        \rho_{\mathrm{lab},f}
        =U_s\rho_0U_s^\dagger,
        \qquad R_s\in SO(3).
      \]
      Thomas--BMT determines $R_s$ along the trajectory.

    \column{0.47\textwidth}
      \textbf{2. The downstream beam defines new axes}
      \[
        \rho_{b,f}
        =U_o^\dagger\rho_{\mathrm{lab},f}U_o.
      \]
      $R_o$ follows the transported momentum and defines the local $x_b,y_b,z_b$ frame.
  \end{columns}

  \vspace{0.30em}
  \[
    \boxed{R_{\mathrm{rel}}=R_o^{\mathsf T}R_s},
    \qquad
    \vct p_{b,f}=R_{\mathrm{rel}}\vct p_0,
    \qquad
    \Tens_{b,f}=R_{\mathrm{rel}}\Tens_0R_{\mathrm{rel}}^{\mathsf T}.
  \]

  \vspace{0.10em}
  \conclusion{Lowest order: a common rotation moves the polarization axis, but vector and tensor ranks remain separate.}
  \note{[Sources] Thomas-BMT equation; Ohlsen spin-1 irreducible multipoles; DPOLAR local beam-frame transport convention. The first transformation is physical evolution; the second only re-expresses the same state in the downstream beam basis.}
\end{frame}

% [EN] Show that a distribution of ordinary BMT rotations broadens the tensor sector without generating vector polarization. / [CN] 说明普通 BMT 旋转的分布只会扩展张量子空间，而不会产生矢量极化。
\begin{frame}{BMT field spread cannot convert tensor to vector}
  Each phase-space point $\xi=(x,x',y,y',\Delta p/p,\ldots)$ has its own calculated rotation:
  \[
    \vct p_f(\xi)=R_\xi\vct p_0,
    \qquad
    \Tens_f(\xi)=R_\xi\Tens_0R_\xi^{\mathsf T}.
  \]

  The polarimeter measures the ensemble average:
  \[
    \overline{\vct p}_f=\int d\xi\,f(\xi)R_\xi\vct p_0,
    \qquad
    \overline{\Tens}_f=\int d\xi\,f(\xi)R_\xi\Tens_0R_\xi^{\mathsf T}.
  \]

  \vspace{0.10em}
  \begin{columns}[T,onlytextwidth]
    \column{0.46\textwidth}
      If $\vct p_0=0$, then exactly
      \[
        \overline{\vct p}_f=0.
      \]
    \column{0.48\textwidth}
      The averaged tensor can become biaxial and require all
      \[
        \boxed{5\ \text{tensor components}}.
      \]
  \end{columns}

  \vspace{0.20em}
  \conclusion{Different $\vct B(\xi)$ under linear BMT can open up to 5 tensor degrees; it cannot give tensor$\rightarrow$vector conversion.}
  \note{[Sources] Wigner rotation law for spin-1 irreducible multipoles; DPOLAR phase-space-averaged BMT transport. A convex average of rank-preserving rotations remains rank preserving.}
\end{frame}

% [EN] Add the distinct higher-order mechanisms that can couple vector and tensor multipoles and therefore open the full density matrix. / [CN] 单独加入能够耦合矢量与张量多极矩、从而打开完整密度矩阵的高阶机制。
\begin{frame}{Rank mixing opens all 8 state parameters}
  Tensor--vector conversion requires spin dynamics beyond the linear BMT term. For example,
  \[
    H_{\rm spin}
    =-\mu\,\vct S\!\cdot\!\vct B
     -\beta_T\bigl(\vct S\!\cdot\!\vct B\bigr)^2+\cdots,
    \qquad
    \overline\rho_f=\int d\xi\,f(\xi)\,\mathcal E_\xi[\rho_0].
  \]

  \begin{columns}[T,onlytextwidth]
    \column{0.47\textwidth}
      \textbf{Quadratic spin interaction}

      Tensor magnetic/electric polarizability or other quadratic spin couplings can generate
      \[
        \vct p_0=0,\ \Tens_0\neq0
        \quad\longrightarrow\quad
        \vct p_f\neq0.
      \]
    \column{0.47\textwidth}
      \textbf{Spin-dependent loss or material interaction}

      Spin-dependent loss, material interactions, collimation, and acceptance can change both ranks and the density-matrix eigenvalues.
  \end{columns}

  \vspace{0.35em}
  \conclusion{Once rank mixing is allowed, the fit must allow the full $8=3_{\rm vector}+5_{\rm tensor}$ parameters.}
  \note{[Sources] A. J. Silenko, Phys. Rev. C 77, 021001 (2008), arXiv:0711.2390; general spin-1 density-matrix evolution. Silenko gives an explicit initially tensor-polarized deuteron beam acquiring vector polarization through tensor magnetic polarizability.}
\end{frame}

% [EN] Derive the one-station blind directions directly from the measured d-p scattering cross section. / [CN] 直接从实测 d-p 散射截面推导单台极化计的盲方向。
\begin{frame}{The cross section shows the two blind directions}
  \footnotesize
  \begin{align*}
    \frac{\sigma(\theta,\phi)}{\sigma_0(\theta)}={}&1
      +\frac32\left(p_x\sin\phi+p_y\cos\phi\right)A_y(\theta)\\
      &+\frac23\left(p_{xz}\cos\phi-p_{yz}\sin\phi\right)A_{xz}(\theta)\\
      &+\frac16\left[(p_{xx}-p_{yy})\cos2\phi-2p_{xy}\sin2\phi\right]
        \left[A_{xx}(\theta)-A_{yy}(\theta)\right]\\
      &+\frac12p_{zz}A_{zz}(\theta).
  \end{align*}
  \normalsize

  \vspace{-0.35em}
  \begin{columns}[T,onlytextwidth]
    \column{0.47\textwidth}
      \textbf{$p_z$ is absent}\\[0.15em]
      The $d$--$p$ reaction has no longitudinal-vector term in this geometry.
    \column{0.47\textwidth}
      \textbf{$p_{xy}$ vanishes at the four arms}\\[0.15em]
      At L/R/U/D, $\phi=0^\circ,90^\circ,180^\circ,270^\circ$, so $\sin2\phi=0$.
  \end{columns}

  \vspace{0.20em}
  {\color{Teal}\bfseries One calibrated multi-angle station measures 6/8 density-matrix parameters, or 4/5 tensor parameters.}
  \note{[Sources] DPOLAR Cartesian d-p scattering response. Multiple polar-angle groups separate py from pxz and px from pyz through their distinct Ay and Axz dependences.}
\end{frame}

% [EN] Show explicitly how the characterized BigRIPS rotation maps both one-station blind directions into measured channels. / [CN] 明确展示已表征的 BigRIPS 转动如何把单台测量的两个盲方向映射到可测通道。

  \]

  \begin{columns}[T,onlytextwidth]
    \column{0.47\textwidth}
      \textbf{Vector blind direction}
      \begin{align*}
        p'_x&=c\,p_x+s\,p_z,\\
        p'_z&=-s\,p_x+c\,p_z.
      \end{align*}
      Hence an upstream $p_z$ gives
      \[
        \boxed{p'_x=0.461951\,p_z},

    \column{0.47\textwidth}
      \textbf{Tensor blind direction}
      \begin{align*}
        p'_{xy}&=c\,p_{xy}+s\,p_{yz},\\
        p'_{yz}&=-s\,p_{xy}+c\,p_{yz}.
      \end{align*}
      Hence an upstream $p_{xy}$ gives
      \[
        \boxed{p'_{yz}=-0.461951\,p_{xy}},
      \]
      which is measured at station 2.
  \end{columns}

  \vspace{0.10em}
  {\color{Teal}\bfseries Two stations: tensor $5/5$; full density matrix $8/8$.}
  \note{[Sources] DPOLAR BigDpol/T11-to-F13 central-ray transport and joint-rank analysis.}
\end{frame}
